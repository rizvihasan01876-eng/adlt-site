SECURITY

Realtime Database rules

Replace YOUR_ADMIN_UID with your administrator’s Firebase Authentication UID in these rules. Public visitors can only query published videos / enabled categories and increment a published video’s view count by one.

{
  "rules": {
    "videos": {
      ".indexOn": ["status"],
      ".read": "(auth != null && auth.uid == 'YOUR_ADMIN_UID') || (query.orderByChild == 'status' && query.equalTo == 'published')",
      "$videoId": {
        ".write": "auth != null && auth.uid == 'YOUR_ADMIN_UID'",
        "views": {
          ".read": "data.parent().child('status').val() == 'published' || (auth != null && auth.uid == 'YOUR_ADMIN_UID')",
          ".write": "data.parent().child('status').val() == 'published' && newData.isNumber() && newData.val() == (data.val() == null ? 1 : data.val() + 1)",
          ".validate": "newData.isNumber()"
        }
      }
    },
    "categories": {
      ".indexOn": ["status"],
      ".read": "(auth != null && auth.uid == 'YOUR_ADMIN_UID') || (query.orderByChild == 'status' && query.equalTo == 'enabled')",
      "$categoryId": {
        ".write": "auth != null && auth.uid == 'YOUR_ADMIN_UID'"
      }
    },
    "settings": {
      ".read": true,
      ".write": "auth != null && auth.uid == 'YOUR_ADMIN_UID'"
    },
    "connectionSignals": {
      "$uid": {
        ".read": "auth != null && auth.uid == $uid",
        ".write": "auth != null && auth.uid == $uid"
      }
    }
  }
}

Paste this JSON into Realtime Database → Rules after replacing the UID. Keep the views child rule: it permits only a single +1 transaction on published videos; all other video and category writes require the admin UID. The demo connection signal is written under connectionSignals/{admin UID}. Add the project’s web config and Realtime Database URL in script.js.
